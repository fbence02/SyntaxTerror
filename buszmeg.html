Szennyezettség: ${stop.rating}`); }); } async function fetchBusStops() { try { // Ide jön majd az igazi API végpont const response = await fetch('https://sajat-api.hu/api/megallok'); if (!response.ok) throw new Error(`Szerverhiba: ${response.status}`); const data = await response.json(); renderBusStops(data); } catch (error) { console.error('Hiba az adatok betöltésekor:', error); } } // 5.
